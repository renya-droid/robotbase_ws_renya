cmake_minimum_required(VERSION 3.8)
project(robotbase_costmap_plugins)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_library(hokuyo_obstacle_layer SHARED src/hokuyo_obstacle_layer.cpp)
target_compile_features(hokuyo_obstacle_layer PUBLIC cxx_std_17)
target_include_directories(hokuyo_obstacle_layer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(hokuyo_obstacle_layer nav2_costmap_2d pluginlib rclcpp std_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS hokuyo_obstacle_layer LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_hokuyo_layer test/test_hokuyo_layer.cpp)
  target_link_libraries(test_hokuyo_layer hokuyo_obstacle_layer)
  ament_target_dependencies(test_hokuyo_layer nav2_costmap_2d pluginlib rclcpp std_msgs)
endif()
ament_package()
