cmake_minimum_required(VERSION 3.8)
project(robotbase_bt_nodes)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_util REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
add_library(robotbase_remove_traversed_goals_bt_node SHARED src/remove_traversed_goals.cpp)
target_compile_definitions(robotbase_remove_traversed_goals_bt_node PRIVATE BT_PLUGIN_EXPORT)
target_include_directories(robotbase_remove_traversed_goals_bt_node PRIVATE include)
ament_target_dependencies(robotbase_remove_traversed_goals_bt_node
  behaviortree_cpp nav2_behavior_tree nav2_util tf2_ros rclcpp geometry_msgs)
install(TARGETS robotbase_remove_traversed_goals_bt_node LIBRARY DESTINATION lib)
ament_export_libraries(robotbase_remove_traversed_goals_bt_node)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_passage test/test_passage.cpp)
  target_include_directories(test_passage PRIVATE include)
endif()
ament_package()
