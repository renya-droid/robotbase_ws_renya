cmake_minimum_required(VERSION 3.8)
project(external_camera_people_layer)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
add_library(${PROJECT_NAME} SHARED src/people_layer.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}
  nav2_costmap_2d nav2_util geometry_msgs pluginlib rclcpp rcl_interfaces)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY verification DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/manual_people.py scripts/verify_sim.py DESTINATION lib/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(nav2_costmap_2d nav2_util geometry_msgs pluginlib rclcpp rcl_interfaces)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(nav2_navfn_planner REQUIRED)
  ament_add_gtest(test_people_layer test/test_people_layer.cpp)
  target_link_libraries(test_people_layer ${PROJECT_NAME})
  ament_target_dependencies(test_people_layer nav2_navfn_planner)
endif()
ament_package()
